import os
from flask import Flask, request, jsonify
from slack_sdk import WebClient
from slack_sdk.errors import SlackApiError
import openai
from dotenv import load_dotenv

load_dotenv()

app = Flask(__name__)

slack_token = os.getenv('SLACK_BOT_TOKEN')
client = WebClient(token=slack_token)
openai.api_key = os.getenv('OPENAI_API_KEY')

def send_message(channel, text):
    try:
        response = client.chat_postMessage(channel=channel, text=text)
        return response
    except SlackApiError as e:
        return f'Error: {e.response["error"]}'

def get_openai_response(prompt):
    try:
        response = openai.Completion.create(
            engine="text-davinci-003",
            prompt=prompt,
            max_tokens=150
        )
        return response.choices[0].text.strip()
    except Exception as e:
        return f"Error: {str(e)}"

def vibe_check(message):
    if "vibe" in message.lower():
        return "Vibe check successful! 👍"
    return "Vibe check failed. 😔"

@app.route("/vibecheck", methods=["POST"])
def slack_event():
    data = request.get_json()
    if "challenge" in data:
        return jsonify({'challenge': data["challenge"]})  # for URL verification
    if "event" in data:
        user_message = data["event"].get("text", "")
        channel = data["event"].get("channel", "")
        if user_message and channel:
            response_text = vibe_check(user_message)
            send_message(channel, response_text)
    return "OK", 200

if __name__ == "__main__":
    app.run(port=3000)
